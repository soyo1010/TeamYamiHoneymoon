Google Sheets 양방향 연동 (GAS)

웹앱에서 선택한 숙소·투어·투표 내역이 구글 시트에 실시간으로 저장됩니다.

* 웹앱에서 변경/투표한 내역은 State 시트에, 구글 시트 Options 탭에 직접 행을 추가한 후보 옵션은 프론트엔드 카드로 자동 동기화됩니다.

📋 Google Sheets `Options` 시트 컬럼 스키마 (직접 행 추가 시 참고)
id | stageId | category | title | subtitle | priceKrw | priceUsd | rating | badge | pros | cons | url | notes

• stageId: FLIGHT, D0315, D0316, D0320, D0321, D0322, D0323, D0324, D0325
• category: 항공 · 숙소 · 프로그램 · 이동수단 · priceKrw는 2인 총액(KRW) 기준입니다.

파트너 이름 설정 (투표 버튼 표시용)

설치 방법 (1분 완료):

  1. 새 구글 스프레드시트를 열고 상단 메뉴 확장 프로그램 > Apps Script 클릭
  2. 아래 코드를 복사해 붙여넣고 저장 후 우측 상단 배포 > 새 배포 클릭
  3. 유형: 웹 앱 / 액세스 권한: 모든 사용자(Anyone) 선택 후 배포
  4. 생성된 웹 앱 URL을 복사해 [1. 웹앱 URL 연결] 탭에 붙여넣기

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const optSheet = ensureOptionsSheet(ss);
  const optRows = optSheet.getDataRange().getValues();
  const options = [];
  if (optRows.length > 1) {
    const headers = optRows[0].map(h => String(h).trim());
    optRows.slice(1).forEach(r => {
      if (!r[0] && !r[3]) return;
      const obj = {};
      headers.forEach((h, idx) => {
        obj[h] = r[idx] !== undefined ? r[idx] : '';
      });
      obj.priceKrw = Number(String(obj.priceKrw).replace(/[^0-9.-]/g, '')) || 0;
      options.push(obj);
    });
  }

  let state = null;
  const stateSheet = ss.getSheetByName(STATE_SHEET);
  if (stateSheet && stateSheet.getLastRow() >= 2) {
    const raw = stateSheet.getRange(2, 1).getValue();
    if (raw) {
      try { state = JSON.parse(raw); } catch (err) { state = null; }
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    options: options,
    state: state
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const now = new Date().toLocaleString('ko-KR');

    if (payload.state) {
      let sSheet = ss.getSheetByName(STATE_SHEET);
      if (!sSheet) {
        sSheet = ss.insertSheet(STATE_SHEET);
        sSheet.getRange(1, 1, 1, 2).setValues([['payloadJson', 'updatedAt']]).setFontWeight('bold');
      }
      sSheet.getRange(2, 1, 1, 2).setValues([[JSON.stringify(payload.state), now]]);
    }

    if (Array.isArray(payload.options)) {
      const oSheet = ensureOptionsSheet(ss);
      const existingRows = oSheet.getDataRange().getValues();
      const existingById = {};
      if (existingRows.length > 1) {
        const hRow = existingRows[0].map(h => String(h).trim());
        const idIdx = hRow.indexOf('id');
        existingRows.slice(1).forEach(row => {
          const rowId = idIdx >= 0 ? String(row[idIdx] || '').trim() : '';
          if (rowId) existingById[rowId] = row;
        });
      }

      const outRows = [OPTIONS_HEADERS];
      const seenIds = new Set();

      payload.options.forEach(item => {
        const id = String(item.id || '').trim();
        if (id) seenIds.add(id);
        const prev = id && existingById[id] ? existingById[id] : null;
        outRows.push([
          id || ('opt-' + Utilities.getUuid().slice(0, 8)),
          item.stageId || item.dayId || (prev ? prev[1] : 'D0315'),
          item.category || (prev ? prev[2] : '숙소'),
          item.title || (prev ? prev[3] : ''),
          item.subtitle || (prev ? prev[4] : ''),
          Number(item.priceKrw !== undefined ? item.priceKrw : (item.priceKRW || 0)) || 0,
          Number(item.priceUsd) || (prev ? Number(prev[6]) || 0 : 0),
          item.rating || (prev ? prev[7] : ''),
          item.badge || (prev ? prev[8] : ''),
          item.pros || (prev ? prev[9] : ''),
          item.cons || (prev ? prev[10] : ''),
          item.url || item.link || (prev ? prev[11] : ''),
          item.notes || item.memo || (prev ? prev[12] : '')
        ]);
      });

      // Preserve any manual rows added in Sheets if not present in payload
      Object.keys(existingById).forEach(rowId => {
        if (!seenIds.has(rowId)) {
          outRows.push(existingById[rowId]);
        }
      });

      oSheet.clearContents();
      oSheet.getRange(1, 1, outRows.length, OPTIONS_HEADERS.length).setValues(outRows);
      oSheet.getRange(1, 1, 1, OPTIONS_HEADERS.length).setFontWeight('bold').setBackground('#F3F4F6');
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', updatedAt: now }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
