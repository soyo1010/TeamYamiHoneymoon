파트너 이름 설정 (투표 버튼 표시용)

설치 방법 (1분 완료):

  1. 새 구글 스프레드시트를 열고 상단 메뉴 확장 프로그램 > Apps Script 클릭
  2. 아래 코드를 복사해 붙여넣고 저장 후 우측 상단 배포 > 새 배포 클릭
  3. 유형: 웹 앱 / 액세스 권한: 모든 사용자(Anyone) 선택 후 배포
  4. 생성된 웹 앱 URL을 복사해 [1. 웹앱 URL 연결] 탭에 붙여넣기
const SHEET_NAME = 'HoneymoonPlan';

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'empty', items: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'empty', items: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  const headers = rows[0];
  const items = rows.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => {
      if (h === 'selected' || h === 'voteA' || h === 'voteB') {
        obj[h] = row[i] === true || row[i] === 'TRUE';
      } else if (h === 'priceKRW') {
        obj[h] = Number(row[i]) || 0;
      } else {
        obj[h] = row[i];
      }
    });
    return obj;
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', items }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const items = payload.items || [];
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
    }
    sheet.clearContents();
    const headers = ['id', 'dayId', 'category', 'title', 'subtitle', 'priceKRW', 'selected', 'pros', 'cons', 'link', 'voteA', 'voteB', 'memo', 'updatedAt'];
    const rows = [headers];
    const now = new Date().toLocaleString('ko-KR');
    items.forEach(item => {
      rows.push([
        item.id || '',
        item.dayId || '',
        item.category || '',
        item.title || '',
        item.subtitle || '',
        Number(item.priceKRW) || 0,
        Boolean(item.selected),
        item.pros || '',
        item.cons || '',
        item.link || '',
        Boolean(item.voteA),
        Boolean(item.voteB),
        item.memo || '',
        now
      ]);
    });
    sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#F3F4F6');
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: items.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
